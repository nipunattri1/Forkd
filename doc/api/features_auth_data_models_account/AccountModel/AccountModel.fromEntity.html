factory AccountModel.fromEntity(AccountEntity entity) {
  return switch (entity) {
    GithubAccountEntity(
      :final tokenHash,
      :final login,
      :final avatarUrl,
      :final domain,
    ) =>
      AccountModel.github(
        tokenHash: tokenHash,
        login: login,
        avatarUrl: avatarUrl,
        domain: domain,
      ),
    GiteeAccountEntity(
      :final tokenHash,
      :final login,
      :final avatarUrl,
      :final domain,
    ) =>
      AccountModel.gitee(
        tokenHash: tokenHash,
        login: login,
        avatarUrl: avatarUrl,
        domain: domain,
      ),
    GitlabAccountEntity(
      :final tokenHash,
      :final username,
      :final avatarUrl,
      :final gitlabId,
      :final domain,
      :final name,
      :final accessLevel,
      :final bio,
      :final clientId,
    ) =>
      AccountModel.gitlab(
        tokenHash: tokenHash,
        username: username,
        avatarUrl: avatarUrl,
        gitlabId: gitlabId,
        domain: domain,
        name: name,
        accessLevel: accessLevel,
        bio: bio,
        clientId: clientId,
      ),
  };
}
